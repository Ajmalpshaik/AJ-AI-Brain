#

#
#   python -m venv venv
#   venv\Scripts\python.exe -m pip install -r requirements.txt
#
# IMPORTANT on a company-managed machine: redirect pip's temp folder first, so
# no package DLLs are unpacked into %TEMP% where antivirus will scan them:
#
#   set TEMP=<this folder>\pip-temp
#   set TMP=<this folder>\pip-temp
